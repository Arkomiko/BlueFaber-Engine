package bluefaber.generated;

import net.fabricmc.api.ModInitializer;
import net.fabricmc.fabric.api.biome.v1.BiomeModifications;
import net.fabricmc.fabric.api.biome.v1.BiomeSelectors;
import net.minecraft.block.AbstractBlock;
import net.minecraft.block.Block;
import net.minecraft.item.BlockItem;
import net.minecraft.item.Item;
import net.minecraft.registry.Registries;
import net.minecraft.registry.Registry;
import net.minecraft.registry.RegistryKey;
import net.minecraft.registry.RegistryKeys;
import net.minecraft.util.Identifier;
import net.minecraft.world.gen.GenerationStep;

public final class ContentMod implements ModInitializer {
    private static final String NAMESPACE = "@NAMESPACE@";

    @Override
    public void onInitialize() {
        // The comments after generated operations point back to their graph nodes.
@REGISTRATION@
    }

    private static void registerBlock(String name, float hardness, boolean requiresTool) {
        var settings = AbstractBlock.Settings.create().strength(hardness);
        // Tool tags control break speed and tier. This flag prevents drops from unsuitable tools.
        if (requiresTool) settings.requiresTool();
        var id = Identifier.of(NAMESPACE, name);
        var block = Registry.register(Registries.BLOCK, id, new Block(settings));
        Registry.register(Registries.ITEM, id, new BlockItem(block, new Item.Settings()));
    }

    private static void registerItem(String name) {
        Registry.register(Registries.ITEM, Identifier.of(NAMESPACE, name), new Item(new Item.Settings()));
    }

    private static void addOverworldOre(String name) {
        var feature = RegistryKey.of(RegistryKeys.PLACED_FEATURE, Identifier.of(NAMESPACE, name));
        BiomeModifications.addFeature(BiomeSelectors.foundInOverworld(), GenerationStep.Feature.UNDERGROUND_ORES, feature);
    }
}
